-- 1. สร้างตารางประชากร (citizens) CREATE TABLE public.citizens ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, house_number TEXT NOT NULL, fullname TEXT NOT NULL, welfare_group TEXT DEFAULT 'ประชาชนทั่วไป', image_url TEXT, created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); -- 2. เปิดใช้งาน Row Level Security (RLS) ALTER TABLE public.citizens ENABLE ROW LEVEL SECURITY; -- 3. สร้าง Policy อนุญาตให้อ่านข้อมูลได้ทุกคน (Public Read) CREATE POLICY "Allow Public Read Citizens" ON public.citizens FOR SELECT USING (true); -- 4. สร้าง Policy อนุญาตให้เพิ่ม/แก้ไข/ลบ ข้อมูลได้ CREATE POLICY "Allow Public All Operations Citizens" ON public.citizens FOR ALL USING (true) WITH CHECK (true);
